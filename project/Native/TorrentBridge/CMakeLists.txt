cmake_minimum_required(VERSION 3.20)
project(AsanTorrent LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
if(POLICY CMP0167)
  cmake_policy(SET CMP0167 OLD)
endif()
if(NOT ASAN_LIBTORRENT_SOURCE OR NOT ASAN_BOOST_SOURCE)
  message(FATAL_ERROR "Supply pinned ASAN_LIBTORRENT_SOURCE and ASAN_BOOST_SOURCE")
endif()
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(build_tests OFF CACHE BOOL "" FORCE)
set(build_examples OFF CACHE BOOL "" FORCE)
set(build_tools OFF CACHE BOOL "" FORCE)
set(python-bindings OFF CACHE BOOL "" FORCE)
set(deprecated-functions OFF CACHE BOOL "" FORCE)
set(encryption OFF CACHE BOOL "" FORCE)
set(logging OFF CACHE BOOL "" FORCE)
set(CMAKE_DISABLE_FIND_PACKAGE_OpenSSL TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_GnuTLS TRUE)
set(CMAKE_DISABLE_FIND_PACKAGE_LibGcrypt TRUE)
set(Boost_NO_BOOST_CMAKE ON)
set(Boost_NO_SYSTEM_PATHS ON)
set(BOOST_ROOT "${ASAN_BOOST_SOURCE}")
set(BOOST_INCLUDEDIR "${ASAN_BOOST_SOURCE}")
# tvOS disallows writable mmap; CommonCrypto supplies SHA, not TLS transport.
add_compile_definitions(TORRENT_HAVE_MMAP=0 TORRENT_USE_COMMONCRYPTO=1)
add_subdirectory("${ASAN_LIBTORRENT_SOURCE}" libtorrent EXCLUDE_FROM_ALL)
add_library(AsanTorrentBridge STATIC AsanTorrent.cpp)
target_include_directories(AsanTorrentBridge PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/include")
target_link_libraries(AsanTorrentBridge PUBLIC torrent-rasterbar)
set_target_properties(AsanTorrentBridge PROPERTIES XCODE_ATTRIBUTE_CODE_SIGNING_ALLOWED NO)
